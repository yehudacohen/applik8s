```ts title="src/guestbook.ts"
const Entry = application.model(entries, { database: Database });

Entry.on.create(async created => {
  await Notifications.send({
    topic: 'guestbook',
    message: created.value.message,
  });
});

application.profile('local', profile => {
  profile.provide(Database, Database.postgres());
  profile.provide(Notifications, Notifications.console());
});
```

## One application, explicit guarantees

Write application semantics in ordinary typed code. Applik8s discovers the graph, resolves concrete
implementations, explains lifecycle and authority, and generates the runtime boundaries needed by the
selected profile.
